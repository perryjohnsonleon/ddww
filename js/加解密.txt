function encode(str) {
  return btoa(encodeURIComponent(str).replace(
    /%([0-9A-F]{2})/g,
    (_, hex) => String.fromCharCode(parseInt(hex, 16))
  ));
}

function decode(str) {
  return decodeURIComponent(
    atob(str).split('').map(
      ch => '%' + ch.charCodeAt(0).toString(16).padStart(2, '0')
    ).join('')
  );
}
